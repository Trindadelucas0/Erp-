# Deploy na VPS — executar na pasta do projeto (ex.: ~/PROJETOS/Erp-)
# Ordem importa: migração ANTES do build.

cd ~/PROJETOS/Erp-

git pull

npm install
cd frontend && npm install && cd ..

npm run db:migrate:deploy
npm run db:generate

npm run build

pm2 restart ecosystem.config.cjs

pm2 restart erp-api erp-web

pm2 restart all
