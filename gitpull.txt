# Deploy na VPS — executar SEMPRE na raiz do projeto (ex.: ~/PROJETOS/Erp-)
# NÃO rode estes comandos de dentro de frontend/ — db:* e o build completo
# ficam no package.json da raiz.
# Ordem importa: migração ANTES do build.

cd ~/PROJETOS/Erp-

# Confirme: pwd deve terminar em Erp- (não em Erp-/frontend)
pwd

git pull

npm install
cd frontend && npm install && cd ..

npm run db:migrate:deploy
npm run db:generate

npm run build


pm2 restart ecosystem.config.cjs

pm2 restart erp-api erp-web

pm2 restart all
